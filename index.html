// INICIALIZACIÓN BLINDADA: BLOQUEA EL LOGIN HASTA DESCARGAR LA NUBE async function initFirebase() { const bar = document.getElementById('splashBar'); const statusTxt = document.getElementById('syncStatusText'); if(statusTxt) statusTxt.innerText = "Conectando con la nube global..."; const openLoginScreen = () => { if(bar) bar.style.width = '100%'; setTimeout(() => { const splash = document.getElementById('splashScreen'); if(splash) { splash.classList.add('fade-out-splash'); setTimeout(() => { splash.style.display = 'none'; const login = document.getElementById('loginScreen'); if(login) login.classList.remove('hidden'); }, 500); } }, 300); }; try { app = initializeApp(firebaseConfig); db = getFirestore(app); auth = getAuth(app); await signInAnonymously(auth); cloudUser = auth.currentUser; isCloudConnected = true; if(statusTxt) statusTxt.innerText = "Sincronizando datos con el servidor..."; const docRef = doc(db, 'jvolt_taller_data', 'master_state'); // ESCUCHA ACTIVA: Espera obligatoriamente a traer los datos de la nube antes de dejar entrar onSnapshot(docRef, (docSnap) => { if (docSnap.exists()) { const cloudData = docSnap.data(); if (cloudData && cloudData.state) { window.appState = cloudData.state; localStorage.setItem('jvolt_master_state', JSON.stringify(window.appState)); if (window.currentUser) { renderApp(); } } } else { setDoc(docRef, { state: window.appState }); } if(statusTxt) statusTxt.innerText = "¡Sincronización Cloud establecida!"; openLoginScreen(); }, (error) => { console.debug("Error de red Firebase:", error); if(statusTxt) statusTxt.innerText = "Modo Local Activo"; openLoginScreen(); }); } catch (e) { console.debug("Fallo de conexión:", e); if(statusTxt) statusTxt.innerText = "Iniciando modo offline..."; openLoginScreen(); } }
