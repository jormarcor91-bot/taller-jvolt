// INICIALIZACIÓN CON PURGA AUTOMÁTICA DE CACHÉ PARA MÓVILES async function initFirebase() { let splashCompleted = false; // Control de versión estricto para evitar que los celulares retengan datos viejos const APP_VERSION = "v2.5_force_sync"; if (localStorage.getItem('jvolt_app_ver') !== APP_VERSION) { localStorage.clear(); localStorage.setItem('jvolt_app_ver', APP_VERSION); } const finishSplash = () => { if (splashCompleted) return; splashCompleted = true; const bar = document.getElementById('splashBar'); if(bar) bar.style.width = '100%'; setTimeout(() => { const splash = document.getElementById('splashScreen'); if(splash) { splash.classList.add('fade-out-splash'); setTimeout(() => { splash.style.display = 'none'; const login = document.getElementById('loginScreen'); if(login) login.classList.remove('hidden'); }, 500); } }, 300); }; setTimeout(finishSplash, 2000); try { app = initializeApp(firebaseConfig); db = getFirestore(app); auth = getAuth(app); await signInAnonymously(auth); cloudUser = auth.currentUser; isCloudConnected = true; const statusTxt = document.getElementById('syncStatusText'); if(statusTxt) statusTxt.innerText = "Sincronización Cloud Global Activa"; const docRef = doc(db, 'jvolt_taller_data', 'master_state'); onSnapshot(docRef, (docSnap) => { if (docSnap.exists()) { const cloudData = docSnap.data(); if (cloudData && cloudData.state) { window.appState = cloudData.state; localStorage.setItem('jvolt_master_state', JSON.stringify(window.appState)); if (window.currentUser) { renderApp(); } } } else { setDoc(docRef, { state: window.appState }); } finishSplash(); }, (error) => { console.debug("Firestore snapshot note:", error); finishSplash(); }); } catch (e) { console.debug("Firebase init fallback mode:", e); finishSplash(); } }
